local function SaveCF()
    local success = pcall(function()
        local character = plr.Character
        if not character then
            character = plr.CharacterAdded:Wait(1)
        end
        if character then
            local root = character:WaitForChild("HumanoidRootPart", 1)
            if root then
                savedCFrame = root.CFrame
            end
        end
    end)
    return savedCFrame ~= nil
end

local function RestoreCF()
    if savedCFrame then
        local character = plr.CharacterAdded:Wait()
        task.wait(0.1)
        local root = character:WaitForChild("HumanoidRootPart")
        root.CFrame = savedCFrame
    end
end

local function GetTargetGripArm(targetChar)
    if not targetChar then return nil end
    
    local gripArmNames = {
        "Right Arm", "RightArm", "RightHand", 
        "RightLowerArm", "RightUpperArm",
        "Right Hand", "Right Lower Arm", "Right Upper Arm"
    }
    
    for _, armName in ipairs(gripArmNames) do
        local arm = targetChar:FindFirstChild(armName)
        if arm then
            return arm
        end
    end
    
    return nil
end

local function BreakTargetArm(targetPlayer)
    if targetPlayer and targetPlayer.Character then
        local targetGripArm = GetTargetGripArm(targetPlayer.Character)
        if targetGripArm then
            local currentPos = nil
            if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                currentPos = plr.Character.HumanoidRootPart.CFrame
            end
            
            targetGripArm:BreakJoints()
            
            task.wait(0.05)
            if currentPos and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                plr.Character.HumanoidRootPart.CFrame = currentPos
            end
        end
    end
end

local function ValidateTarget(targetPlayer)
    if not targetPlayer then
        NotifyTargetNotFound()
        return false
    end
    
    if not targetPlayer.Character then
        NotifyPlayerDead()
        return false
    end
    
    local targetHum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not targetHum or targetHum.Health <= 0 then
        NotifyPlayerDead()
        return false
    end
    
    return true
end

local function DetectTargetDeath(targetPlayer, targetName, method)
    task.wait(0.1)
    
    if not targetPlayer then return false end
    
    if not targetPlayer.Character then
        NotifyKillSuccess(targetName, method)
        return true
    end
    
    local targetHum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not targetHum or targetHum.Health <= 0 then
        NotifyKillSuccess(targetName, method)
        return true
    end
    
    local checks = 0
    local maxChecks = 20
    
    repeat
        task.wait(0.1)
        checks = checks + 1
        
        if not targetPlayer then
            NotifyKillSuccess(targetName, method)
            return true
        end
        
        if not targetPlayer.Character then
            NotifyKillSuccess(targetName, method)
            return true
        end
        
        local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            NotifyKillSuccess(targetName, method)
            return true
        end
    until checks >= maxChecks
    
    return false
end

local function KillPlayer(targetPlayer)
    task.spawn(function()
        pcall(function()
            if not ValidateTarget(targetPlayer) then return end
            
            SaveCF()
            
            local targetName2 = targetPlayer.Name
            
            local toolInBackpack = plr.Backpack:FindFirstChildOfClass("Tool")
            local toolEquipped = plr.Character and plr.Character:FindFirstChildOfClass("Tool")
            
            if not toolInBackpack and not toolEquipped then
                NotifyNoTools()
                return
            end
            
            local Tool = toolEquipped or toolInBackpack
            
            if not toolEquipped and toolInBackpack and plr.Character then
                plr.Character.Humanoid:UnequipTools()
                local Humanoid = plr.Character.Humanoid:Clone()
                if plr.Character:FindFirstChild("Animate") then
                    plr.Character.Animate.Disabled = true
                    plr.Character.Humanoid.DisplayDistanceType = "None"
                end
                plr.Character.Humanoid:Destroy()
                Humanoid.Parent = plr.Character
                Tool.Parent = plr.Character
            end
            
            local char = plr.Character
            if char and char:FindFirstChildOfClass("Humanoid") then
                char.Humanoid.RequiresNeck = false
                char.Humanoid.BreakJointsOnDeath = true
                char.Humanoid.PlatformStand = true
                char.Humanoid:TakeDamage(9999)
                char.Humanoid:ChangeState(15)
                char.HumanoidRootPart.Anchored = false
            end
            local LocalPlayer = plr
            
            BreakTargetArm(targetPlayer)
            
            local function DoTouch(targetPlayer, tool)
                if not targetPlayer or not targetPlayer.Character then return end
                if not tool or not tool:FindFirstChild("Handle") then return end
                
                pcall(function()
                    firetouchinterest(targetPlayer.Character.HumanoidRootPart, tool.Handle, 0)
                    firetouchinterest(targetPlayer.Character.HumanoidRootPart, tool.Handle, 1)
                end)
                
                pcall(function()
                    local targetArm = targetPlayer.Character:FindFirstChild("Right Hand") or 
                                     targetPlayer.Character:FindFirstChild("Right Arm")
                    if targetArm then
                        local original = tool.Handle.CFrame
                        tool.Handle.CFrame = targetArm.CFrame
                        task.wait(0.01)
                        tool.Handle.CFrame = original
                    end
                end)
            end
            
            local flag = false
            task.defer(function()
                if Tool and Tool:FindFirstChild("Handle") then
                    Tool.Handle.AncestryChanged:Wait()
                    flag = true
                end
            end)
            
            repeat
                if targetPlayer then
                    DoTouch(targetPlayer, Tool)
                end
                task.wait(0.1)
            until flag
            
            task.spawn(function()
                DetectTargetDeath(targetPlayer, targetName2, "Kill")
            end)
            
            task.delay(0.1, function()
                pcall(function()
                    if char then
                        char:BreakJoints()
                        
                        wait(0.1)
                        if char:FindFirstChildOfClass("Humanoid") then
                            for _, child in pairs(char.Humanoid:GetChildren()) do
                                pcall(function() child:Remove() end)
                            end
                        end
                        wait(0.1)
                        
                        if plr.Character then
                            plr.Character.Humanoid.MaxHealth = 0
                            plr.Character.Humanoid.Health = 0

                            plr.Character:BreakJoints()
                            plr.Character.Humanoid.BreakJointsOnDeath = true

                            wait(0.5)

                            plr.Character:Remove()
                            plr.Character:Destroy()
                        end
                        
                        plr.Character = nil
                        LocalPlayer.CharacterAdded:Wait()
                        task.wait(0.1)
                        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and savedCFrame then
                            LocalPlayer.Character.HumanoidRootPart.CFrame = savedCFrame
                        end
                    end
                end)
            end)
        end)
    end)
end

local function BanishPlayer(targetPlayer)
    task.spawn(function()
        pcall(function()
            if not ValidateTarget(targetPlayer) then return end
            
            SaveCF()
            
            local targetName2 = targetPlayer.Name
            local targetChar = targetPlayer.Character
            
            local toolInBackpack = plr.Backpack:FindFirstChildOfClass("Tool")
            local toolEquipped = plr.Character and plr.Character:FindFirstChildOfClass("Tool")
            
            if not toolInBackpack and not toolEquipped then
                NotifyNoTools()
                return
            end
            
            local plrChar = plr.Character
            if not plrChar then return end
            
            if plrChar:FindFirstChildOfClass("Humanoid") then
                local humClone = plrChar.Humanoid:Clone()
                humClone.Parent = plrChar
                humClone.Name = "Hum"
                plrChar.Humanoid:Destroy()
            end
            
            for _, tool in pairs(plr:GetDescendants()) do
                if tool.ClassName == "Tool" and tool.Parent ~= plrChar then
                    tool.Parent = plrChar
                end
            end
            
            local tool = plrChar:FindFirstChildOfClass("Tool")
            if tool and tool:FindFirstChild("Handle") and targetChar then
                local targetArm = targetChar:FindFirstChild("Right Hand") or targetChar:FindFirstChild("Right Arm")
                local originalPos = tool.Handle.CFrame
                
                if targetArm then
                    tool.Handle.CFrame = targetArm.CFrame
                    task.wait(0.1)
                    tool.Handle.CFrame = originalPos
                end
            end
            
            task.wait(0.5)
            
            if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
                targetChar.HumanoidRootPart.CFrame = CFrame.new(0, -100, 0)
            end
            
            if plrChar and plrChar:FindFirstChild("HumanoidRootPart") then
                plrChar.HumanoidRootPart.CFrame = CFrame.new(0, -100, 0)
            end
            
            wait(0.5)
            
            pcall(function()
                plrChar:Remove()
                plrChar:Destroy()
            end)
            
            task.spawn(function()
                DetectTargetDeath(targetPlayer, targetName2, "Banish")
            end)
            
            plr.CharacterAdded:Wait()
            task.wait(0.1)
            RestoreCF()
        end)
    end)
end

local function VoidPlayer(targetPlayer)
    task.spawn(function()
        pcall(function()
            if not ValidateTarget(targetPlayer) then return end
            
            local targetName2 = targetPlayer.Name
            local tchar = targetPlayer.Character
            local thrp = tchar and tchar:FindFirstChild("HumanoidRootPart")
            if not thrp then return end
            
            local char = plr.Character
            if not char then return end
            
            local hum = char:FindFirstChildOfClass("Humanoid")
            local hrp = char:FindFirstChild("HumanoidRootPart")
            
            if not hum or not hrp then return end
            
            local toolInBackpack = plr.Backpack:FindFirstChildOfClass("Tool")
            if not toolInBackpack then
                NotifyNoTools()
                return
            end
            
            local clone = hum:Clone()
            clone.Parent = char
            hum:Destroy()
            
            if toolInBackpack and clone then
                clone:EquipTool(toolInBackpack)
            end
            
            local tool = char:FindFirstChildOfClass("Tool")
            if not tool or not tool:FindFirstChild("Handle") then 
                return
            end
            
            task.wait(0.2)
            
            thrp.CFrame = tool.Handle.CFrame
            
            local targetArm = tchar:FindFirstChild("Right Hand") or 
                             tchar:FindFirstChild("Right Arm") or
                             tchar:FindFirstChild("RightLowerArm")
            
            if targetArm then
                local originalCF = tool.Handle.CFrame
                tool.Handle.CFrame = targetArm.CFrame
                task.wait(0.1)
                tool.Handle.CFrame = originalCF
            end
            
            tool.AncestryChanged:Wait()
            
            task.spawn(function()
                DetectTargetDeath(targetPlayer, targetName2, "Void")
            end)
            
            hrp.CFrame = CFrame.new(Vector3.new(0, workspace.FallenPartsDestroyHeight + 1, 0))
            wait(0.2)
            
            pcall(function()
                char:Remove()
                char:Destroy()
            end)
        end)
    end)
end

RandomButton.MouseButton1Click:Connect(function()
    local players = game:GetService("Players"):GetPlayers()
    local otrosJugadores = {}
    for _, v in pairs(players) do
        if v ~= plr then
            table.insert(otrosJugadores, v)
        end
    end
    if #otrosJugadores > 0 then
        local randomPlayer = otrosJugadores[math.random(1, #otrosJugadores)]
        SearchBox.Text = randomPlayer.Name
    end
end)
